# =========================================================
# CORAZÓN
# =========================================================

def crear_corazon(x, y, tamaño, color="#d00032"):

    puntos = []

    for i in range(0, 361, 3):

        t = math.radians(i)

        px = 16 * math.sin(t) ** 3
        py = (
            13 * math.cos(t)
            - 5 * math.cos(2*t)
            - 2 * math.cos(3*t)
            - math.cos(4*t)
        
        puntos.append(
            (
                x + px * tamaño / 16,
                y - py * tamaño / 16
            )
        )

    canvas.create_polygon(
        puntos,
        fill=color,
        outline=""
    )

# =========================================================
# FLOR
# =========================================================

def crear_flor(x, y, tamaño=1):

    color = random.choice([
        "#d00032",
        "#e0003b",
        "#ff1744",
        "#b8002d"
    ])

    # TALLO

    canvas.create_line(
        x,
        y + 40*tamaño,
        x,
        y + 230*tamaño,
        fill="#258a38",
        width=int(7*tamaño)
    )

    # HOJAS

    canvas.create_oval(
        x - 65*tamaño,
        y + 130*tamaño,
        x,
        y + 165*tamaño,
        fill="#319447",
        outline=""
    )

    canvas.create_oval(
        x,
        y + 165*tamaño,
        x + 65*tamaño,
        y + 200*tamaño,
        fill="#287e39",
        outline=""
    )

    # PÉTALOS

    posiciones = [
        (0, -35),
        (32, -12),
        (28, 25),
        (0, 40),
        (-28, 25),
        (-32, -12)
    ]

    for px, py in posiciones:

        canvas.create_oval(
            x + (px-35)*tamaño,
            y + (py-35)*tamaño,
            x + (px+35)*tamaño,
            y + (py+35)*tamaño,
            fill=color,
            outline="#980027",
            width=2
        )

    # CENTRO

    canvas.create_oval(
        x - 23*tamaño,
        y - 23*tamaño,
        x + 23*tamaño,
        y + 23*tamaño,
        fill="#ffd21c",
        outline="#a87500",
        width=3
    )


# =========================================================
# RAMO
# =========================================================

def crear_ramo(cx, cy, escala=1):

    posiciones = [
        (-95, 0),
        (-45, -45),
        (0, -65),
        (45, -45),
        (95, 0),
        (-65, 55),
        (0, 40),
        (65, 55)
    ]

    for x, y in posiciones:

        crear_flor(
            cx + x*escala,
            cy + y*escala,
            escala
        )

    # PAPEL

    canvas.create_polygon(
        cx - 150*escala,
        cy + 190*escala,

        cx + 150*escala,
        cy + 190*escala,

        cx + 90*escala,
        cy + 350*escala,

        cx - 90*escala,
        cy + 350*escala,

        fill="#ed9db5",
        outline="#c96c8a",
        width=3
    )

    # CINTA

    canvas.create_rectangle(
        cx - 20*escala,
        cy + 250*escala,
        cx + 20*escala,
        cy + 320*escala,
        fill="#c90035",
        outline=""
    )

# MUCHOS CORAZONES
# =========================================================

corazones = []

def crear_corazones():

    for i in range(25):

        x = random.randint(30, 970)
        y = random.randint(100, 700)

        tamaño = random.randint(8, 18)

        corazones.append({
            "x": x,
            "y": y,
            "t": tamaño,
            "vel": random.uniform(0.5, 1.8)
        })


def animar_corazones():

    if not animando:
        return

    # Crear corazones nuevamente
    crear_corazones()

    for c in corazones:

        c["y"] -= c["vel"]

        crear_corazon(
            c["x"],
            c["y"],
            c["t"],
            "#d00032"
        )

    if animando:

        ventana.after(
            100,
            animar_corazones
        )
